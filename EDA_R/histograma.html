5 Histograma

Al pasar el mouse sobre cada barra se muestra el rango de magnitud y la frecuencia.

p_hist <- ggplot(sismos, aes(x = magnitud)) +
  geom_histogram(bins = 30, fill = "steelblue", color = "white") +
  geom_vline(xintercept = resumen_estadistico$media, color = "red", linetype = "dashed") +
  labs(title = "Histograma de magnitudes sísmicas",
       x = "Magnitud", y = "Frecuencia") +
  theme_minimal()

ggplotly(p_hist, tooltip = c("x", "y")) %>%
  layout(hovermode = "closest")
