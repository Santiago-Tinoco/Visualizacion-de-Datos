4 Tabla resumen de estadísticas descriptivas

Estadísticas sobre la variable magnitud: tendencia central, dispersión, forma de la distribución y rango intercuartílico.

resumen_estadistico <- sismos %>%
  summarise(
    n             = n(),
    media         = mean(magnitud, na.rm = TRUE),
    mediana       = median(magnitud, na.rm = TRUE),
    moda          = as.numeric(names(sort(table(magnitud), decreasing = TRUE)[1])),
    desv_std      = sd(magnitud, na.rm = TRUE),
    varianza      = var(magnitud, na.rm = TRUE),
    minimo        = min(magnitud, na.rm = TRUE),
    maximo        = max(magnitud, na.rm = TRUE),
    rango         = maximo - minimo,
    q1            = quantile(magnitud, 0.25, na.rm = TRUE),
    q3            = quantile(magnitud, 0.75, na.rm = TRUE),
    iqr           = IQR(magnitud, na.rm = TRUE),
    cv_pct        = (desv_std / media) * 100,          # coeficiente de variación
    asimetria     = skewness(magnitud, na.rm = TRUE),  # skewness
    curtosis      = kurtosis(magnitud, na.rm = TRUE),  # kurtosis
    err_std_media = desv_std / sqrt(n)
  )

4.1 Rango intercuartílico y detección de outliers

Usando la regla de Tukey (1.5 × IQR) para identificar valores atípicos de magnitud.

lim_inf <- resumen_estadistico$q1 - 1.5 * resumen_estadistico$iqr
lim_sup <- resumen_estadistico$q3 + 1.5 * resumen_estadistico$iqr

outliers <- sismos %>% filter(magnitud < lim_inf | magnitud > lim_sup)

cat("Límite inferior:", round(lim_inf, 2), "| Límite superior:", round(lim_sup, 2), "\n")
## Límite inferior: 4.35 | Límite superior: 6.35
cat("Número de outliers detectados (regla 1.5*IQR):", nrow(outliers), "\n")
## Número de outliers detectados (regla 1.5*IQR): 35
