8 Diagrama de Pareto

8.1 Por región

place de USGS suele ser muy granular (ej. “10km NE of Ridgecrest, CA”); si tu definición de región es más amplia, agrupa primero con str_detect() o expresiones regulares.

pareto_region <- sismos %>%
  count(region, name = "frecuencia") %>%
  arrange(desc(frecuencia)) %>%
  slice_head(n = 10) %>%
  mutate(
    region = factor(region, levels = region),  # conserva el orden descendente
    porcentaje = frecuencia / sum(frecuencia) * 100,
    porcentaje_acum = cumsum(porcentaje)
  )

plot_ly(pareto_region) %>%
  add_bars(
    x = ~region, y = ~frecuencia, name = "Frecuencia",
    marker = list(color = "steelblue"),
    hovertemplate = "%{x}<br>Frecuencia: %{y}<extra></extra>"
  ) %>%
  add_lines(
    x = ~region, y = ~porcentaje_acum, name = "% acumulado", yaxis = "y2",
    line = list(color = "red", width = 2),
    marker = list(color = "red"),
    hovertemplate = "%{x}<br>Acumulado: %{y:.1f}%<extra></extra>"
  ) %>%
  layout(
    title = "Diagrama de Pareto - Top 10 zonas con más sismos",
    xaxis = list(title = "Zona (place)", tickangle = -45),
    yaxis = list(title = "Frecuencia"),
    yaxis2 = list(title = "% acumulado", overlaying = "y", side = "right",
                   range = c(0, 100)),
    legend = list(x = 0.75, y = 1.15, orientation = "h"),
    margin = list(b = 120)
  )

8.2 Por rango de magnitud

pareto_data <- sismos %>%
  mutate(rango_mag = cut(magnitud,
                          breaks = c(-Inf, 3, 4, 5, 6, 7, Inf),
                          labels = c("<3","3-4","4-5","5-6","6-7","7+"))) %>%
  count(rango_mag, name = "frecuencia") %>%
  arrange(desc(frecuencia)) %>%
  mutate(
    rango_mag = factor(rango_mag, levels = rango_mag),
    porcentaje = frecuencia / sum(frecuencia) * 100,
    porcentaje_acum = cumsum(porcentaje)
  )

plot_ly(pareto_data) %>%
  add_bars(
    x = ~rango_mag, y = ~frecuencia, name = "Frecuencia",
    marker = list(color = "steelblue"),
    hovertemplate = "Magnitud %{x}<br>Frecuencia: %{y}<extra></extra>"
  ) %>%
  add_lines(
    x = ~rango_mag, y = ~porcentaje_acum, name = "% acumulado", yaxis = "y2",
    line = list(color = "red", width = 2),
    marker = list(color = "red"),
    hovertemplate = "Magnitud %{x}<br>Acumulado: %{y:.1f}%<extra></extra>"
  ) %>%
  layout(
    title = "Diagrama de Pareto - Sismos por rango de magnitud",
    xaxis = list(title = "Rango de magnitud"),
    yaxis = list(title = "Frecuencia"),
    yaxis2 = list(title = "% acumulado", overlaying = "y", side = "right",
                   range = c(0, 100)),
    legend = list(x = 0.75, y = 1.15, orientation = "h")
  )
