def factorial(n: int) -> int:
    """Calcula el factorial del número dado"""
    if n == 0:
            return 1
    else:
        return n * factorial(n - 1)

num = int(input("Ingrese un número entero: "))

print(f"El factorial de {num} es: {factorial(num)}")

  1. Crear matriz de ceros: Genere una matriz AAA de dimensiones (m×n)(m \times n)(m×n) inicializada con ceros.

  2. Llenar matriz con números aleatorios (usando while):

    • Utilizando un bucle while, reemplace cada columna de la matriz AAA por vectores de números aleatorios enteros en el rango [−1,3][-1, 3][−1,3]

    • Recorra columna por columna

  3. Transformar elementos (usando while):

    • Utilizando dos bucles while anidados, recorra todos los elementos de la matriz

    • Reemplace los elementos negativos por -1

    • Reemplace los elementos no negativos (cero o positivos) por 5

  4. Repetir transformación (usando for):

    • Realice el mismo ejercicio del paso 3, pero esta vez utilizando dos bucles for anidados

    • Compare los resultados para verificar que ambos métodos producen la misma matriz

import numpy as np

m = int(input("Ingrese el número de filas de la matriz: "))
n = int(input("Ingrese el número de columnas de la matriz: "))

#1
A = np.zeros((m, n), dtype=int)

#2
col = 0
while col < A.shape[1]:
    A[:, col] = np.random.randint(-1, 4, size=A.shape[0])
    col += 1
print(f"La matriz generada de ceros es:\n{A}", end="\n\n")

#3
matriz_1 = A.copy()
col = 0
fil = 0
while col < matriz_1.shape[1]:
    while fil < matriz_1.shape[0]:
        if matriz_1[fil, col] < 0:
            matriz_1[fil, col] = -1
        else:
            matriz_1[fil, col] = 5
        fil += 1
    fil = 0
    col += 1

print(f"La matriz usando el ciclo while es:\n{matriz_1}", end="\n\n")

#4
matriz_2 = A.copy()
for i in range(matriz_2.shape[0]):
    for j in range(matriz_2.shape[1]):
        if matriz_2[i, j] == -1:
            matriz_2[i, j] = -1
        else:
            matriz_2[i, j] = 5

print(f"La matriz usando el ciclo for es:\n{matriz_2}", end="\n\n")
print(f"Comparando las matrices resultantes de los métodos 3 y 4, son iguales?: {np.array_equal(matriz_1, matriz_2)}")
